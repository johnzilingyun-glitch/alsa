```python
# input_sanitizer.py
class InputSanitizer:
    INJECTION_PATTERNS = [
        r"ignore\s+(previous|all)\s+(instructions?|prompts?)",
        r"you\s+are\s+now\s+",
        r"system\s*:\s*",
        r"<\|im_start\|>",
    ]

    def sanitize(self, text: str) -> str:
        # 移除潜在注入载荷
        for pattern in self.INJECTION_PATTERNS:
            text = re.sub(pattern, "[FILTERED]", text, flags=re.IGNORECASE)
        # 限制长度
        if len(text) > 500:
            text = text[:500]
        return text

    def sanitize_stock_name(self, name: str) -> str:
        # 仅允许中英文、数字、常见符号
        return re.sub(r'[^\w\s\u4e00-\u9fff().\-]', '', name)
```

```python
# agent_memory.py
class AgentMemory:
    def __init__(self, lancedb_store, sqlite_session):
        self.vector_store = lancedb_store  # LanceDB 用于语义检索
        self.db = sqlite_session            # SQLite 用于精确查询

    async def recall(self, symbol: str, role: str, query: str) -> list[dict]:
        """检索相关历史分析"""
        # 1. 向量检索: 找到语义相似的历史分析
        vector_results = self.vector_store.search(symbol, query, limit=3)
        # 2. 精确检索: 找到同股票的历史分析
        exact_results = self.db.query(
            "SELECT * FROM agent_memory WHERE symbol=? AND role=? ORDER BY created_at DESC LIMIT 5",
            [symbol, role]
        )
        return self._merge_results(vector_results, exact_results)

    async def store(self, symbol: str, role: str, analysis: str, outcome: str = None):
        """存储分析结果"""
        # 向量化存入 LanceDB
        self.vector_store.upsert_documents([{
            "doc_id": f"{symbol}_{role}_{datetime.now().isoformat()}",
            "symbol": symbol,
            "text": analysis[:2000],
            "source_type": "agent_analysis",
        }])
        # 存入 SQLite
        self.db.execute(
            "INSERT INTO agent_memory (symbol, role, analysis, outcome) VALUES (?, ?, ?, ?)",
            [symbol, role, analysis[:5000], outcome]
        )
```

```python
# backtest_engine_v2.py
class BacktestEngineV2:
    def __init__(self):
        self.commission_rate = 0.001     # 佣金 0.1%
        self.stamp_duty_rate = 0.001     # 印花税 0.1% (卖出)
        self.slippage_rate = 0.002       # 滑点 0.2%

    def run(self, symbols, start_date, end_date, initial_capital=1000000):
        # 1. 获取数据
        prices = self._fetch_prices(symbols, start_date, end_date)
        # 2. 计算协方差（Ledoit-Wolf 收缩估计）
        returns = prices.pct_change().dropna()
        cov_shrunk = self._ledoit_wolf_shrink(returns)
        # 3. 计算 GMV 权重
        weights = self._gmv_weights(cov_shrunk)
        # 4. 模拟交易（含交易成本）
        portfolio_value = self._simulate_trades(prices, weights, initial_capital)
        # 5. 计算指标
        metrics = self._compute_metrics(portfolio_value)
        # 6. Walk-forward validation
        oos_metrics = self._walk_forward(prices, initial_capital)
        return {
            "in_sample": metrics,
            "out_of_sample": oos_metrics,
            "weights": weights,
        }

    def _ledoit_wolf_shrink(self, returns):
        """Ledoit-Wolf 协方差收缩估计"""
        # 简化实现: shrink toward diagonal
        sample_cov = returns.cov()
        n = sample_cov.shape[0]
        trace = np.trace(sample_cov.values)
        shrink_target = (trace / n) * np.eye(n)
        alpha = 0.2  # 收缩强度
        return sample_cov * (1 - alpha) + shrink_target * alpha

    def _simulate_trades(self, prices, weights, initial_capital):
        """含交易成本的交易模拟"""
        portfolio = initial_capital
        positions = {}
        history = []
        for date in prices.index:
            # 计算目标持仓
            target_value = {s: portfolio * weights[s] for s in weights}
            # 计算交易成本
            cost = self._compute_trading_cost(positions, target_value, prices.loc[date])
            portfolio -= cost
            # 更新持仓价值
            portfolio = sum(target_value.values())
            history.append({"date": date, "value": portfolio})
        return pd.DataFrame(history)
```

```python
# risk_models.py
class RiskModels:
    def var_historical(self, returns, confidence=0.95, holding_period=1):
        """Historical Simulation VaR"""
        sorted_returns = np.sort(returns)
        index = int((1 - confidence) * len(sorted_returns))
        return -sorted_returns[index] * np.sqrt(holding_period)

    def cvar(self, returns, confidence=0.95):
        """Conditional VaR (Expected Shortfall)"""
        var = self.var_historical(returns, confidence)
        return -np.mean(returns[returns <= -var])

    def kelly_half(self, win_rate, avg_win, avg_loss):
        """Half-Kelly with Bayesian adjustment"""
        kelly = win_rate - (1 - win_rate) / (avg_win / avg_loss)
        return max(0, kelly * 0.5)  # Half-Kelly

    def max_drawdown(self, equity_curve):
        """最大回撤"""
        peak = equity_curve.expanding(min_periods=1).max()
        drawdown = (equity_curve - peak) / peak
        return drawdown.min()
```
